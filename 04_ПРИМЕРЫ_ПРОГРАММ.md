# Книга 4: Готовые программы на MX

Лучший способ выучить язык — читать и запускать готовый код. Собрал сюда несколко программ
от простого к сложному. Каждую можно скопировать в файл, запустить и докрутить под себя.

Запуск любого примера: сохрани как `пример.mx`, потом `mx пример.mx пример.mxb --run`.
Работает на любой ОС где стоит MX.

## Проверка числа на простоту

```mx
func простое(n: int): bool {
    if n < 2 { return false }
    for (let i: int = 2; i * i <= n; i = i + 1) {
        if n % i == 0 { return false }
    }
    return true
}

func main(): int {
    for (let i: int = 1; i <= 50; i = i + 1) {
        if простое(i) {
            System.println(Convert.toStr(i) + " — простое число")
        }
    }
    return 0
}
```

## Счётчик слов в тексте

```mx
func main(): int {
    let текст = "кот собака кот птица кот"
    let слова = String.split(текст, " ")

    let счёт: Map[str, int] = Map[str, int].new()

    for (let слово in слова) {
        let сколько = Map[str, int].getOr(счёт, слово, 0)
        Map[str, int].set(счёт, слово, сколько + 1)
    }

    let ключи = Map[str, int].keys(счёт)
    for (let ключ in ключи) {
        System.println(ключ + " : " + Convert.toStr(Map[str, int].get(счёт, ключ)))
    }
    return 0
}
```

## Простой веб-сервер

```mx
func main(): int {
    let сервер = HttpServer.create(8080)
    System.println("Сервер на порту 8080...")

    loop {
        let запрос = сервер.accept()
        if запрос.valid == false { continue }

        if String.equals(запрос.path(), "/") {
            сервер.replyHtml(запрос, 200, "<h1>Привет из MX!</h1>")
        } else if String.equals(запрос.path(), "/привет") {
            let имя = запрос.query("имя")
            сервер.replyText(запрос, 200, "Привет, " + имя + "!")
        } else {
            сервер.notFound(запрос)
        }
    }
    return 0
}
```

Открой браузер, зайди на `http://localhost:8080`. Сервер будет работать пока не закроешь программу.

## Прыгающий мяч

```mx
func main(): int {
    let окно = Gfx.open(640, 480, "Мяч")

    let x: int = 320
    var y: int = 100
    var скорость: int = 0
    let гравитация: int = 2
    let пол: int = 420

    while Gfx.isOpen(окно) {
        Gfx.clear(окно, 0, 0, 0)

        скорость = скорость + гравитация
        y = y + скорость
        if y > пол {
            y = пол
            скорость = скорость - 30   // отскок
        }

        Gfx.fillCircle(окно, x, y, 30, 255, 100, 100)
        Gfx.text(окно, 10, 10, "Высота: " + Convert.toStr(y))

        let событие = Gfx.pollEvent(окно)
        if событие == 3 { break }

        Gfx.show(окно)
        System.sleep(16)   // ~60 кадров в секунду
    }

    Gfx.close(окно)
    return 0
}
```

## Аргументы командной строки

```mx
func main(): int {
    let аргументы = Args.all()

    if CliKit.has(аргументы, "--help") {
        System.println("  --имя <значение>   твоё имя")
        System.println("  --громко           громкий режим")
        return 0
    }

    let имя = CliKit.value(аргументы, "--имя", "Гость")
    let громко = CliKit.has(аргументы, "--громко")

    if громко {
        System.println("ПРИВЕТ, " + String.toUpper(имя) + "!!!")
    } else {
        System.println("Привет, " + имя)
    }
    return 0
}
```

Запуск: `mx пример.mx пример.mxb --run --имя Саша --громко`

## Обработка ошибок

```mx
func делить(a: int, b: int): int {
    if b == 0 {
        panic("Нельзя делить на ноль")
    }
    return a / b
}

func main(): int {
    try {
        let результат = делить(10, 2)
        System.println("10 / 2 = " + Convert.toStr(результат))

        let плохо = делить(10, 0)   // ошибка
    } catch (сообщение) {
        System.println("Ошибка: " + сообщение)
    }
    return 0
}
```

## Советы

- Начни с малого: сначала «Привет, мир», потом добавь переменые, потом циклы.
- Имена переменных можно давать на русском — MX понимает.
- Читай сообщения об ошибках: там файл, строка и место. Обычно это мелоч — забыл `:`,
  пропустил `{`.
- Для отладки используй `System.println` — печатай значения и смотри.
- Что-то не нашёл — загляни в Книгу 3, там список готовых функций.
